### Update sendCommand in VoiceUI Component

Replace the existing `sendCommand` function (lines 210-291) with:

```typescript
const sendCommand = async () => {
  if (!transcript) return;
  
  setPhase('thinking');
  setShowTranscript(false);
  addHistory(transcript, 'voice');
  
  // Lock scroll during processing
  document.body.style.overflow = 'hidden';
  
  // Orchestrate command with intent detection
  try {
    const result = await orchestrateCommand(transcript);
    console.log('[VoiceUI] Orchestration result:', result);
    
    // Add request with detected intent type and metadata
    const requestId = addRequest(transcript, result.intent.type, result.intent);
    updateRequestStatus(requestId, 'Processing');
    
    // If workflow was triggered (not fallback to stream)
    if (!result.fallbackToStream && result.workflowResponse) {
      setPhase('responding');
      
      // Show workflow response message
      const workflowMessage = result.workflowResponse.message;
      addResponse(workflowMessage);
      
      setTimeout(() => {
        updateRequestStatus(requestId, 'Complete');
        setPhase('idle');
        setBarHeights(Array(BAR_COUNT).fill(BASE_HEIGHT));
        setAmplitude(0);
        setTranscript('');
        setShowTranscript(false);
        
        // Restore scroll and pointer events
        document.body.style.overflow = 'auto';
        document.body.style.pointerEvents = 'auto';
      }, 2000);
      return;
    }
    
    // Fallback to AI streaming
    setPhase('responding');
    
    let responseText = '';
    
    streamCleanupRef.current = streamAIResponse(
      transcript,
      // onChunk
      (chunk) => {
        responseText += chunk;
        addResponse(responseText);
      },
      // onComplete
      () => {
        console.log('[VoiceUI] Stream completed, restoring UI state');
        updateRequestStatus(requestId, 'Complete');
        setPhase('idle');
        setBarHeights(Array(BAR_COUNT).fill(BASE_HEIGHT));
        setAmplitude(0);
        setTranscript('');
        setShowTranscript(false);
        
        // Restore scroll and pointer events
        document.body.style.overflow = 'auto';
        document.body.style.pointerEvents = 'auto';
        
        // Clear cleanup ref
        streamCleanupRef.current = null;
      },
      // onError
      (error) => {
        console.error('[VoiceUI] AI streaming error:', error);
        updateRequestStatus(requestId, 'Error', error.message);
        
        // Restore scroll immediately on error
        document.body.style.overflow = 'auto';
        document.body.style.pointerEvents = 'auto';
        
        // Fallback to mock response
        const fallbackResponse = `I'm processing your request about "${transcript}". Let me help you with that.`;
        
        addResponse(fallbackResponse);
        setTimeout(() => {
          updateRequestStatus(requestId, 'Complete');
          setPhase('idle');
          setBarHeights(Array(BAR_COUNT).fill(BASE_HEIGHT));
          setAmplitude(0);
          setTranscript('');
          setShowTranscript(false);
          
          // Clear cleanup ref
          streamCleanupRef.current = null;
        }, 1000);
      }
    );
  } catch (error) {
    console.error('[VoiceUI] Orchestration error:', error);
    
    // If orchestration fails entirely, create generic request
    const requestId = addRequest(transcript, 'GENERIC');
    updateRequestStatus(requestId, 'Error', 'Orchestration failed');
    
    // Restore UI
    setPhase('idle');
    document.body.style.overflow = 'auto';
    document.body.style.pointerEvents = 'auto';
  }
};
```

const handleSend = async () => {
  if (!input.trim() || isStreaming) return;

  const command = input.trim();
  setInput('');
  setIsStreaming(true);
  setStreamingText('');
  
  // Lock scroll during processing
  document.body.style.overflow = 'hidden';
  
  try {
    // Orchestrate command with intent detection
    const result = await orchestrateCommand(command);
    console.log('[CommandCenter] Orchestration result:', result);
    
    // Add request with detected intent type and metadata
    const requestId = addRequest(command, result.intent.type, result.intent);
    addHistory(command, 'text');
    
    setTimeout(() => {
      updateRequestStatus(requestId, 'Processing');
      
      // If workflow was triggered
      if (!result.fallbackToStream && result.workflowResponse) {
        const workflowMessage = result.workflowResponse.message;
        setStreamingText(workflowMessage);
        addResponse(workflowMessage);
        
        setTimeout(() => {
          updateRequestStatus(requestId, 'Complete');
          setIsStreaming(false);
          document.body.style.overflow = 'auto';
          document.body.style.pointerEvents = 'auto';
        }, 2000);
        return;
      }
      
      // Fallback to AI streaming
      let responseText = '';
      
      streamCleanupRef.current = streamAIResponse(
        command,
        // onChunk
        (chunk) => {
          responseText += chunk;
          setStreamingText(responseText);
        },
        // onComplete
        () => {
          console.log('[CommandCenter] Stream completed, restoring UI state');
          addResponse(responseText);
          updateRequestStatus(requestId, 'Complete');
          setIsStreaming(false);
          
          // Restore scroll and pointer events
          document.body.style.overflow = 'auto';
          document.body.style.pointerEvents = 'auto';
          
          // Clear cleanup ref

        },
        // onError

          console.error('[CommandCenter] AI streaming error:', error);
          updateRequestStatus(requestId, 'Error', error.message);

## Key Points

1. **Intent Detection Happens First:** Before creating any request, we detect the intent using `orchestrateCommand()`

2. **Request Creation with Type:** When adding a request, we now pass the detected intent type and metadata:
   ```typescript
   addRequest(transcript, result.intent.type, result.intent)
   ```

3. **Workflow vs Streaming:** 
   - If `result.fallbackToStream` is `false`, workflow APIs were called successfully
   - If `true`, we use the existing AI streaming mechanism

4. **Error Handling:** All errors gracefully fallback to generic behavior

5. **Console Logging:** `[Orchestrator]`, `[Intent]`, `[Workflow]` prefixes help with debugging
